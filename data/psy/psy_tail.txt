    let c: Vec<char> = c_input
        .split(',')

fn count_chars(s: &str, c: &[char]) -> Vec<(char, usize)> {
    // Initialize a hashmap to keep track of the number of occurrences of each character
    let mut counts = std::collections::HashMap::new();
    for char in c {
        counts.insert(*char, 0);
    }

    // Loop through each character in the string
    for char in s.chars() {
        // If the character matches one of the input characters, increment the counter for that character
        if let Some(count) = counts.get_mut(&char.to_ascii_lowercase()) {
            *count += 1;
        }
    }

    // Convert hashmap to vector, sort by frequency, and return
    let mut result: Vec<(char, usize)> = counts.into_iter().collect();
    result.sort_by(|a, b| b.1.cmp(&a.1));
    result
}
